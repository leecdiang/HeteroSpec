cmake_minimum_required(VERSION 3.20)
project(rtx244_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to a pinned complete source tree")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama-build)
add_executable(rtx244_worker native_worker.cpp)
target_link_libraries(rtx244_worker PRIVATE llama)
target_include_directories(rtx244_worker PRIVATE "${LLAMA_SOURCE_DIR}/vendor")
add_executable(rtx244_reference native_reference.cpp)
target_include_directories(rtx244_reference PRIVATE "${LLAMA_SOURCE_DIR}/vendor")
target_link_libraries(rtx244_reference PRIVATE llama)
