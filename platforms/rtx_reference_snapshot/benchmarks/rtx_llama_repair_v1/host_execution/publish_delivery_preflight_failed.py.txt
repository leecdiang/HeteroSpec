"""Publish the same-private-repository release after complete verified delivery."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import subprocess


def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for block in iter(lambda:f.read(4*1024**2),b''):h.update(block)
    return h.hexdigest()
def read(p):return json.loads(p.read_text(encoding='utf-8'))


def main():
    p=argparse.ArgumentParser()
    for name in ('archive','transport','notes','out'):p.add_argument('--'+name,type=Path,required=True)
    for name in ('tag','snapshot'):p.add_argument('--'+name,required=True)
    p.add_argument('--release-id',type=int,required=True);a=p.parse_args();a.out.mkdir(exist_ok=False)
    env=dict(os.environ,GODEBUG='http2client=0');repo='leecdiang/HeteroSpec'
    def command(name,args):
        r=subprocess.run(['gh',*map(str,args)],capture_output=True,text=True,encoding='utf-8',env=env,timeout=600)
        with (a.out/(name+'.json')).open('x',encoding='utf-8') as f:json.dump({'argv':['gh',*map(str,args)],'exit_code':r.returncode,'stdout':r.stdout,'stderr':r.stderr},f,indent=2)
        assert r.returncode==0,name+' failed; retained receipt'
        return r.stdout
    raw=a.archive/'RAW_ACCESS.json';index=read(raw);verified=read(a.transport/'remote01/REMOTE_READBACK.json')
    assert verified['status']=='REMOTE_DOWNLOAD_AND_ALL_MEMBERS_READBACK_PASS' and verified['raw_index_sha256']==sha(raw)
    assert read(a.transport/'receipt_remote01/RECEIPT_ASSET_READBACK.json')['status']=='RECEIPT_ASSET_FRESH_REMOTE_READBACK_PASS'
    release=json.loads(command('BEFORE',['api',f'repos/{repo}/releases/{a.release_id}']))
    assert release['draft'] and release['tag_name']==a.tag and release['target_commitish']==a.snapshot
    repository=json.loads(command('REPOSITORY',['api',f'repos/{repo}']));assert repository['private'],'repository visibility changed'
    pages=json.loads(command('ASSETS',['api','--paginate','--slurp',f'repos/{repo}/releases/{a.release_id}/assets?per_page=100']))
    assets=[x for page in pages for x in page];byname={x['name']:x for x in assets};assert len(byname)==len(assets)
    expected=list(index['parts']);delivery=a.transport/'delivery01'
    files=[raw,*[delivery/x['name'] for x in read(delivery/'DELIVERY_ASSETS.json')['assets']],delivery/'DELIVERY_ASSETS.json',a.transport/'delivery_remote01/DELIVERY_METADATA_READBACK.json']
    expected += [{'name:f.name,'bytes':f.stat().st_size,'sha256':sha(f)} for f in files]
    assert set(byname)=={x['name'] for x in expected},'unaccounted or missing remote asset'
    for x in expected:
        asset=byname[x['name']]
        assert asset['state']=='uploaded' and asset['size']==x['bytes'] and asset['digest']=='sha256:'+x['sha256']
    command('PUBLISH',['release','edit',a.tag,'--repo',repo,'--draft=false','--latest=false','--notes-file',a.notes])
    after=json.loads(command('AFTER',['api',f'repos/{repo}/releases/{a.release_id}']))
    assert not after['draft'] and after['tag_name']==a.tag
    fresh=a.out/'fresh';fresh.mkdir()
    command('PUBLISHED_INDEX_READBACK',['release','download',a.tag,'--repo',repo,'--dir',fresh,'--pattern','RAW_ACCESS.json'])
    assert sha(fresh/'RAW_ACCESS.json')==sha(raw)
    result={'status':'PUBLISHED_PRIVATE_RELEASE_AND_AUTHENTICATED_INDEX_READBACK_PASS','release_url':after['html_url'],
      'release_id':a.release_id,'asset_count':len(assets),'raw_index_sha256':sha(raw),'repository_private':True,
      'all_raw_and_metadata_members_previously_read_back':True,'native_requests_added':0,
      'scope':'Repository access controls preserved; authenticated published-index download after complete raw/metadata verification. No merge or manuscript edit.'}
    with (a.out/'PUBLICATION.json').open('x',encoding='utf-8') as f:json.dump(result,f,indent=2)
    print(result['status'],flush=True)


if __name__=='__main__':main()
