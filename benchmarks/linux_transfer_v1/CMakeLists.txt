cmake_minimum_required(VERSION 3.24)
project(reference_compatible_linux LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native Linux only: no Wine or Windows DLL linkage")
endif()
set(ROLE "" CACHE STRING "reference, loop4 or packed")
set(LLAMA_SRC "" CACHE PATH "Frozen full source tree")
set(FRONTEND_SRC "" CACHE PATH "POSIX-adapted frozen frontend")
option(TRANSFER_CUDA "Build actual CUDA numerical backend" ON)
if(NOT ROLE MATCHES "^(reference|loop4|packed)$")
  message(FATAL_ERROR "Explicit role required")
endif()
if(NOT EXISTS "${LLAMA_SRC}/ggml/src/ggml-cuda/CMakeLists.txt")
  message(FATAL_ERROR "Complete numerical source tree required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build all internal numerical modules from source" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "No external backend DLL/SO replacement" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend from same source" FORCE)
set(GGML_CUDA ${TRANSFER_CUDA} CACHE BOOL "CUDA backend from same source" FORCE)
set(LLAMA_USE_SYSTEM_GGML OFF CACHE BOOL "Do not import a system numerical backend" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Explicit portable CPU baseline" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
foreach(opt LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_MTMD LLAMA_OPENSSL LLAMA_BUILD_UI LLAMA_USE_PREBUILT_UI LLAMA_SUBPROCESS)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
# Preserve upstream arithmetic options. Do not add fast-math/FTZ overrides here.
if(TRANSFER_CUDA AND NOT CMAKE_CUDA_ARCHITECTURES MATCHES "^[0-9]+$")
  message(FATAL_ERROR "Use the selected GPU's observed architecture, e.g. 86, not native/all")
endif()
add_subdirectory("${LLAMA_SRC}" llama)
if(ROLE STREQUAL "reference")
  add_executable(transfer-worker native/reference.cpp)
else()
  if(NOT EXISTS "${FRONTEND_SRC}/entry.cpp")
    message(FATAL_ERROR "Adapt the authenticated frontend first")
  endif()
  add_executable(transfer-worker "${FRONTEND_SRC}/entry.cpp")
  target_compile_definitions(transfer-worker PRIVATE PHASE_CANDIDATE=1 HS_SOURCE_VARIANT="original")
  target_include_directories(transfer-worker PRIVATE "${FRONTEND_SRC}" "${CMAKE_BINARY_DIR}/llama/src")
endif()
target_include_directories(transfer-worker PRIVATE native "${LLAMA_SRC}/include" "${LLAMA_SRC}/src" "${LLAMA_SRC}/ggml/include" "${LLAMA_SRC}/vendor" "${LLAMA_SRC}/common")
target_link_libraries(transfer-worker PRIVATE llama ${CMAKE_DL_LIBS})
if(TARGET vendor-hash)
  target_link_libraries(transfer-worker PRIVATE vendor-hash)
endif()
set_target_properties(transfer-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

target_link_options(transfer-worker PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/transfer-worker.map")
