cmake_minimum_required(VERSION 3.20)
project(alignment_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(alignment-metadata metadata.cpp)
target_include_directories(alignment-metadata PRIVATE "${PINNED}/ggml/include" "${PINNED}/vendor" "${GENERATED}" ../../native)
target_link_libraries(alignment-metadata PRIVATE "${BASE_LIB}" psapi)
target_compile_options(alignment-metadata PRIVATE /O2 /W4 /fp:precise)
add_executable(alignment-diagnostic "${GENERATED}/diagnostic.cpp")
target_include_directories(alignment-diagnostic PRIVATE "${PINNED}/ggml/include" "${PINNED}/vendor" "${GENERATED}" ../../native ../../native/host)
target_link_libraries(alignment-diagnostic PRIVATE "${BASE_LIB}" psapi)
target_compile_options(alignment-diagnostic PRIVATE /O2 /W4 /fp:precise)
