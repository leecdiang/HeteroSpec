cmake_minimum_required(VERSION 3.20)
project(b570_prefix_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED LLAMA_SOURCE OR NOT DEFINED HISTORICAL_LIBS)
 message(FATAL_ERROR "Set LLAMA_SOURCE and HISTORICAL_LIBS")
endif()
foreach(lib ggml ggml-base)
 add_library(${lib} SHARED IMPORTED)
 set_target_properties(${lib} PROPERTIES IMPORTED_IMPLIB "${HISTORICAL_LIBS}/runtime/ggml/src/Release/${lib}.lib" IMPORTED_LOCATION "${HISTORICAL_LIBS}/bin/Release/${lib}.dll" INTERFACE_INCLUDE_DIRECTORIES "${LLAMA_SOURCE}/ggml/include")
endforeach()
if(CANDIDATE)
 # Retain PR169 Vulkan and ggml DLLs byte-for-byte. Only relink llama to add
 # the output-free allocator inspector; no numerical source is changed.
 function(llama_add_compile_flags)
 endfunction()
 set(BUILD_SHARED_LIBS ON)
 set(LLAMA_VERSION_BASE "0.4.0")
 set(LLAMA_VERSION_MAJOR 0)
 set(LLAMA_VERSION "0.4.0-dev")
 set(LLAMA_COMMIT "0cae43063")
 add_subdirectory("${LLAMA_SOURCE}/src" runtime/src)
 target_link_libraries(llama PRIVATE ggml-base)
 target_sources(llama PRIVATE "${CMAKE_CURRENT_LIST_DIR}/companion_scope.cpp" "${CMAKE_CURRENT_LIST_DIR}/allocator_probe.cpp")
 target_include_directories(llama PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${LLAMA_SOURCE}/ggml/include")
 target_compile_definitions(llama PRIVATE GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU GGML_USE_VULKAN _CRT_SECURE_NO_WARNINGS)
 target_compile_options(llama PRIVATE /utf-8 /bigobj)
else()
 add_library(llama SHARED IMPORTED)
 set_target_properties(llama PROPERTIES IMPORTED_IMPLIB "${HISTORICAL_LIBS}/runtime/src/Release/llama.lib" IMPORTED_LOCATION "${HISTORICAL_LIBS}/bin/Release/llama.dll")
endif()
add_executable(frontier worker.cpp)
target_include_directories(frontier PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include" "${LLAMA_SOURCE}/vendor")
target_compile_definitions(frontier PRIVATE LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED)
if(CANDIDATE)
 target_compile_definitions(frontier PRIVATE CAND)
endif()
target_compile_options(frontier PRIVATE /fp:precise /O2 /W4 /utf-8)
target_link_libraries(frontier PRIVATE llama ggml ggml-base bcrypt)
