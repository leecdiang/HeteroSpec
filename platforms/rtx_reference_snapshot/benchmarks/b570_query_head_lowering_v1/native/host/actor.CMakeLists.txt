cmake_minimum_required(VERSION 3.20)
project(b570_qh_actor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Vulkan REQUIRED)
file(GLOB LLAMA_SRC "${ACTOR_SOURCE}/*.cpp" "${ACTOR_SOURCE}/models/*.cpp")
add_library(llama-b570-qh SHARED ${LLAMA_SRC})
target_compile_definitions(llama-b570-qh PRIVATE LLAMA_BUILD LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED _CRT_SECURE_NO_WARNINGS)
target_include_directories(llama-b570-qh PRIVATE "${ACTOR_SOURCE}" "${PINNED}/include" "${PINNED}/ggml/include")
target_link_libraries(llama-b570-qh PRIVATE "${LIB}/ggml.lib" "${LIB}/ggml-base.lib")
add_executable(b570-qh-worker "${ACTOR_WORKER}")
target_compile_definitions(b570-qh-worker PRIVATE LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED _CRT_SECURE_NO_WARNINGS)
target_include_directories(b570-qh-worker PRIVATE "${ACTOR_HEADERS}" "${PINNED}/include" "${PINNED}/ggml/include" "${PINNED}/vendor")
target_link_libraries(b570-qh-worker PRIVATE llama-b570-qh "${LIB}/ggml.lib" "${LIB}/ggml-base.lib" Vulkan::Vulkan bcrypt)
foreach(T llama-b570-qh b570-qh-worker)
 target_compile_options(${T} PRIVATE /utf-8 /bigobj)
endforeach()
