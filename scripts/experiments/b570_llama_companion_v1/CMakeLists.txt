cmake_minimum_required(VERSION 3.20)
project(b570_llama_companion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(HS_COMPANION_ORACLE "Unmodified upstream numerical graph; R only" OFF)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "Set LLAMA_SOURCE to a fresh source-stage output")
endif()
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
foreach(feature COMMON TESTS EXAMPLES TOOLS SERVER APP)
 set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_SOURCE}" runtime)
target_sources(llama PRIVATE "${CMAKE_CURRENT_LIST_DIR}/companion_scope.cpp")
target_include_directories(llama PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
add_executable(b570_llama worker.cpp)
target_include_directories(b570_llama PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include" "${LLAMA_SOURCE}/vendor")
target_compile_definitions(b570_llama PRIVATE HS_COMPANION_ORACLE=$<BOOL:${HS_COMPANION_ORACLE}>)
target_link_libraries(b570_llama PRIVATE llama ggml ggml-base)
if(MSVC)
 target_compile_options(b570_llama PRIVATE /O2 /W4 /fp:precise /utf-8)
else()
 target_compile_options(b570_llama PRIVATE -Wall -Wextra -fno-fast-math)
endif()
set_target_properties(b570_llama PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
