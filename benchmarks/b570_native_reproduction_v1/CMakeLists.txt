cmake_minimum_required(VERSION 3.20)
project(b570_native_reproduction LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "Set LLAMA_SOURCE to the prepared source tree")
endif()
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" runtime)
target_include_directories(llama PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native")
foreach(name formal stage_replay)
  add_executable(${name} native/${name}.cpp)
  target_include_directories(${name} PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include" "${LLAMA_SOURCE}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}/native")
  target_compile_options(${name} PRIVATE /O2 /W4 /fp:precise /utf-8)
  target_link_libraries(${name} PRIVATE llama ggml ggml-base)
endforeach()
set_target_properties(formal stage_replay PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
