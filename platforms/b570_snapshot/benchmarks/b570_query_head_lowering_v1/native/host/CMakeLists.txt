cmake_minimum_required(VERSION 3.20)
project(query_head_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(qh-real cpu.cpp)
target_include_directories(qh-real PRIVATE "${PINNED}/ggml/include" "${PINNED}/vendor" ..)
target_link_libraries(qh-real PRIVATE "${BASE_LIB}" psapi)
target_compile_options(qh-real PRIVATE /O2 /W4 /fp:precise)
add_executable(qh-descriptor mock.cpp)
target_include_directories(qh-descriptor PRIVATE ../../tests/mock .. "${PINNED}/vendor")
target_link_libraries(qh-descriptor PRIVATE psapi)
target_compile_options(qh-descriptor PRIVATE /O2 /W4 /fp:precise)
add_executable(qh-vulkan vulkan.cpp)
target_include_directories(qh-vulkan PRIVATE "${PINNED}/ggml/include" "${PINNED}/vendor" ..)
target_link_libraries(qh-vulkan PRIVATE "${BASE_LIB}" psapi)
target_compile_options(qh-vulkan PRIVATE /O2 /W4 /fp:precise)
set_target_properties(qh-vulkan PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/S1")
