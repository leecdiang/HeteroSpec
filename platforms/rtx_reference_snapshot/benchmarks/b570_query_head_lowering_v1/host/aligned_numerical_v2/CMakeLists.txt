cmake_minimum_required(VERSION 3.20)
project(aligned_numerical_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(target aligned-metadata-v2 aligned-host-v2)
  add_executable(${target} host.cpp)
  target_include_directories(${target} PRIVATE "${PINNED}/ggml/include" "${PINNED}/vendor" ../../native)
  target_link_libraries(${target} PRIVATE "${BASE_LIB}" psapi)
  target_compile_options(${target} PRIVATE /O2 /W4 /fp:precise)
endforeach()
target_compile_definitions(aligned-metadata-v2 PRIVATE PANEL_CPU_AUDIT=1)
