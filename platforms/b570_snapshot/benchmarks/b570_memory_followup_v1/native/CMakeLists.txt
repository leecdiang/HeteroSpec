cmake_minimum_required(VERSION 3.20)
project(b570_memory_followup LANGUAGES CXX)
find_package(Vulkan REQUIRED)
set(PINNED_SOURCE "${CMAKE_SOURCE_DIR}/../dependencies/pinned-source")
set(PINNED_BUILD "${CMAKE_SOURCE_DIR}/../dependencies/pinned-build")
file(GLOB CORE "candidate-src/*.cpp" "candidate-src/models/*.cpp")
add_library(llama-b570-memory SHARED ${CORE})
target_compile_features(llama-b570-memory PRIVATE cxx_std_17)
target_compile_options(llama-b570-memory PRIVATE /utf-8 /bigobj)
target_compile_definitions(llama-b570-memory PRIVATE LLAMA_BUILD LLAMA_SHARED _CRT_SECURE_NO_WARNINGS GGML_USE_CPU GGML_USE_VULKAN GGML_SHARED GGML_BACKEND_SHARED)
target_include_directories(llama-b570-memory PRIVATE candidate-src "${PINNED_SOURCE}/include" "${PINNED_SOURCE}/ggml/include")
target_link_libraries(llama-b570-memory PRIVATE "${PINNED_BUILD}/ggml/src/Release/ggml.lib" "${PINNED_BUILD}/ggml/src/Release/ggml-base.lib")
add_executable(b570-memory-worker worker.cpp)
target_compile_features(b570-memory-worker PRIVATE cxx_std_17)
target_compile_definitions(b570-memory-worker PRIVATE LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED _CRT_SECURE_NO_WARNINGS)
target_include_directories(b570-memory-worker PRIVATE "${PINNED_SOURCE}/include" "${PINNED_SOURCE}/ggml/include" "${PINNED_SOURCE}/vendor")
target_link_libraries(b570-memory-worker PRIVATE llama-b570-memory "${PINNED_BUILD}/ggml/src/Release/ggml.lib" "${PINNED_BUILD}/ggml/src/Release/ggml-base.lib" Vulkan::Vulkan)
