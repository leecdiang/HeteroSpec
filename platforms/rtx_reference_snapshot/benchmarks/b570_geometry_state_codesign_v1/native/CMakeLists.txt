cmake_minimum_required(VERSION 3.20)
project(b570_geometry_state_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Vulkan REQUIRED)
set(HOST "${CMAKE_CURRENT_SOURCE_DIR}/../_host")
set(SRC "${HOST}/source")
set(INC "${HOST}/pr97/dependencies/pinned-source")
set(LIB "${HOST}/pr97/dependencies/pinned-build/ggml/src/Release")
file(GLOB LLAMA_SRC "${SRC}/*.cpp" "${SRC}/models/*.cpp")
add_library(llama-b570-memory SHARED ${LLAMA_SRC})
target_compile_definitions(llama-b570-memory PRIVATE LLAMA_BUILD LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED _CRT_SECURE_NO_WARNINGS)
target_include_directories(llama-b570-memory PRIVATE "${SRC}" "${INC}/include" "${INC}/ggml/include" "${INC}/ggml/src")
target_link_libraries(llama-b570-memory PRIVATE "${LIB}/ggml.lib" "${LIB}/ggml-base.lib")
add_executable(b570-geometry-worker worker.cpp)
target_compile_definitions(b570-geometry-worker PRIVATE LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED _CRT_SECURE_NO_WARNINGS)
target_include_directories(b570-geometry-worker PRIVATE "${INC}/include" "${INC}/ggml/include" "${INC}/vendor")
target_link_libraries(b570-geometry-worker PRIVATE llama-b570-memory "${LIB}/ggml.lib" "${LIB}/ggml-base.lib" Vulkan::Vulkan bcrypt)
add_executable(graph-contract-cpu graph_contract_cpu.cpp)
target_include_directories(graph-contract-cpu PRIVATE "${INC}/ggml/include")
target_compile_definitions(graph-contract-cpu PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
target_link_libraries(graph-contract-cpu PRIVATE "${LIB}/ggml-base.lib")
foreach(T llama-b570-memory b570-geometry-worker graph-contract-cpu)
 target_compile_options(${T} PRIVATE /utf-8 /bigobj)
endforeach()

add_executable(graph-numeric-cpu graph_numeric_cpu.cpp)
target_include_directories(graph-numeric-cpu PRIVATE "${INC}/ggml/include")
target_compile_definitions(graph-numeric-cpu PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
target_compile_options(graph-numeric-cpu PRIVATE /utf-8)
target_link_libraries(graph-numeric-cpu PRIVATE "${LIB}/ggml-base.lib")
